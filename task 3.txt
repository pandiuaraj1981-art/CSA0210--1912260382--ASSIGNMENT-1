#include <stdio.h>

#define RATE1 225
#define RATE2 350
#define RATE3 500
#define FIXED_CHARGE 5000

int main()
{
    int units;
    long long energy_charge;
    long long total;

    printf("Enter units consumed: ");
    if (scanf("%d", &units) != 1 || units < 0)
    {
        printf("Invalid input\n");
        return 1;
    }

    if (units <= 100)
        energy_charge = (long long)units * RATE1;
    else if (units <= 200)
        energy_charge = 100LL * RATE1 +
                        (long long)(units - 100) * RATE2;
    else
        energy_charge = 100LL * RATE1 +
                        100LL * RATE2 +
                        (long long)(units - 200) * RATE3;

    total = energy_charge + FIXED_CHARGE;

    printf("Total Bill: Rs %lld.%02lld\n",
           total / 100, total % 100);

    return 0;
}
