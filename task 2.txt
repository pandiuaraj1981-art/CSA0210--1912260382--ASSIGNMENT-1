#include <stdio.h>

int main()
{
    int bills[] = {4000, 60000, 800000};
    int i;

    long long paise_per_bill = 27500;
    float float_total;
    long long paise_total;
    double double_total;

    printf("Number of Bills\tFloat Total\tPaise Total\n");

    for (i = 0; i < 3; i++)
    {
        float_total = bills[i] * 275.00f;
        paise_total = (long long)bills[i] * paise_per_bill;
        double_total = bills[i] * 275.00;

        printf("%d\t\t%.2f\t%lld\n",
               bills[i], float_total, paise_total);

        printf("Double Total: %.2f\n", double_total);
    }

    return 0;
}
