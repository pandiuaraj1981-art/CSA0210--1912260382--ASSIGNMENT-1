#include <stdio.h>
#define SLAB1 100
#define SLAB2 200
#define RATE1 2.25
#define RATE2 3.50
#define RATE3 5.00
#define FIXED_CHARGE 50.00
int main ()
{
    int previous, current, units;
    double energy charge, total;
printf ("Enter previous reading: ");
    if (scanf ("%d", &previous)! = 1)
        return 1;
  printf ("Enter current reading: ");
    if (scanf ("%d", &current)! = 1)
        return 1;
   if (previous < 0 || previous > 65535 ||
        current < 0 || current > 65535)
    {
        printf ("Invalid meter reading\n");
        return 1;
    }

    if (current >= previous)
        units = current - previous;
    else if (previous == 65535 && current == 0)
        units = 1;
    else
    {
        printf ("Invalid reading: No meter rollover\n");
        return 1;
    }

    if (units <= SLAB1)
        energy charge = units * RATE1;
    else if (units <= SLAB2)
        energy charge = SLAB1 * RATE1 +
                        (units - SLAB1) * RATE2;
    else
        energy charge = SLAB1 * RATE1 +
                        SLAB1 * RATE2 +
                        (units - SLAB2) * RATE3;

    total = energy charge + FIXED_CHARGE;

    printf ("\n--- Electricity Bill ---\n");
    printf ("Units Consumed: %d\n", units);
    printf ("Energy Charge: Rs %.2f\n", energy charge);
    printf ("Fixed Charge: Rs %.2f\n", FIXED_CHARGE);
    printf ("Total Bill: Rs %.2f\n", total);
    return 0;
}
