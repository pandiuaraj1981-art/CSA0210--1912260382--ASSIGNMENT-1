#include <stdio.h>
#define PEAK_RATE 1.20
int main()
{
    int watt_hours = 1000;
    int hour = 19;
    double rate = 5.00;
    double units, bill;

    units = watt_hours / 1000.0;

    if (hour >= 18 && hour < 22)
        rate = rate * PEAK_RATE;

    bill = units * rate;

    printf("Energy Consumed: %.2f kWh\n", units);
    printf("Electricity Bill: Rs %.2f\n", bill);

    return 0;
}
